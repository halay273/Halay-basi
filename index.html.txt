    <script>
        // Web Audio API ile telifsiz ses efektleri (harici dosya gerektirmez)
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            if (type === 'rose') {
                osc.frequency.setValueAtTime(400, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.15);
            } else if (type === 'lion') {
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(150, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.4);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.4);
            } else if (type === 'universe') {
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(300, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.6);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.6);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.6);
            }
        }

        let scoreL = 1240;
        let scoreR = 980;

        function sendGift(emoji, points, target) {
            // Sesi çal
            if(points === 10) playSound('rose');
            else if(points === 100) playSound('lion');
            else playSound('universe');

            // Puanı artır
            if(target === 'left') {
                scoreL += points;
                document.getElementById('scoreLeft').innerText = scoreL;
                createFlyingEmoji(emoji, 100);
            } else {
                scoreR += points;
                document.getElementById('scoreRight').innerText = scoreR;
                createFlyingEmoji(emoji, 300);
            }

            updateBars();
        }

        function updateBars() {
            let total = scoreL + scoreR;
            let percentL = (scoreL / total) * 100;
            let percentR = (scoreR / total) * 100;

            document.getElementById('hpLeft').style.width = percentL + '%';
            document.getElementById('hpRight').style.width = percentR + '%';
        }

        function createFlyingEmoji(emoji, xPos) {
            const container = document.getElementById('mainContainer');
            const el = document.createElement('div');
            el.className = 'flying-gift';
            el.innerText = emoji;
            el.style.left = (xPos + Math.random() * 50) + 'px';
            el.style.bottom = '80px';
            container.appendChild(el);

            setTimeout(() => {
                el.remove();
            }, 1000);
        }

        function updateName() {
            const name = document.getElementById('userInput').value.trim();
            if(name !== "") {
                document.getElementById('leftName').innerText = name;
                document.getElementById('userInput').value = "";
            }
        }

        // Rakibin de arada rastgele hediye alması (canlı yayın heyecanı için)
        setInterval(() => {
            if(Math.random() > 0.6) {
                let pts = Math.random() > 0.5 ? 10 : 100;
                scoreR += pts;
                document.getElementById('scoreRight').innerText = scoreR;
                updateBars();
            }
        }, 3000);
    </script>
